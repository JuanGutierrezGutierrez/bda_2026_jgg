```python
def filtrar_por_provincia(provincia):
    lineas_a_escribir = []
    with client.read('/datalake/renfe/raw/estaciones/estaciones_completo.csv', encoding='latin-1') as reader:
        for linea in reader:
            if linea.split(";")[7].strip('"') == provincia or linea.split(";")[7].strip('"') == "PROVINCIA":
                lineas_a_escribir.append(linea)
    with client.write('/datalake/renfe/processed/estaciones_filtradas.csv', encoding='latin-1', overwrite=True) as writer:
        writer.write("".join(lineas_a_escribir))
```
